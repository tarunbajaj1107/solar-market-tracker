    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.11'
      - name: Install dependencies
        run: pip install requests beautifulsoup4 yfinance pandas
      - name: Execute Scraper
        run: python scraper.py
      - name: Auto-commit updated data

          commit_message: "Auto-update solar pricing benchmarks [skip ci]"
